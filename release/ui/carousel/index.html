Usage

const Vapor = @import("vapor"); const Stack = Vapor.Stack; const Text = Vapor.Text; const Color = Vapor.Types.Color; const Carousel = @import("opaque-ui").Carousel; const Slide = struct { title: []const u8, color: Color }; const slides = [_]Slide{ .{ .title = "Slide One", .color = .hex("#3b82f6") }, .{ .title = "Slide Two", .color = .hex("#8b5cf6") }, .{ .title = "Slide Three", .color = .hex("#ec4899") }, .{ .title = "Slide Four", .color = .hex("#f97316") }, .{ .title = "Slide Five", .color = .hex("#10b981") }, }; // Arrows, swipe and dots come from the carousel; you render the slides. pub fn render() void { Carousel.render(.{ .id = "hero", // required: keeps the index across renders .count = slides.len, .viewport_width = 480, .viewport_height = 280, .stride = 480 + 16, // slide width + gap }).children({ for (slides) |slide| { Stack() .width(.px(480)) .height(.px(280)) .background(slide.color) .radius(.all(12)) .inlineStyle("flex-shrink: 0;", .{}) .layout(.center) .children({ Text(slide.title).font(28, 700, .white).end(); }); } }); }

Options

The variants above: size, looping, arrows and dots.

const Vapor = @import("vapor"); const Stack = Vapor.Stack; const Text = Vapor.Text; const Color = Vapor.Types.Color; const Carousel = @import("opaque-ui").Carousel; const planets = [_][]const u8{ "Mercury", "Venus", "Earth", "Mars" }; const colors = [_]Color{ .hex("#a78bfa"), .hex("#f472b6"), .hex("#38bdf8"), .hex("#fb923c") }; /// One slide of the given size. fn slide(index: usize, width: f32, height: f32) void { Stack() .width(.px(width)) .height(.px(height)) .background(colors[index % colors.len]) .radius(.all(12)) .inlineStyle("flex-shrink: 0;", .{}) .layout(.center) .children({ Text(planets[index]).font(22, 700, .white).end(); }); } /// Smaller slides: the viewport and stride follow the slide size. pub fn compact() void { Carousel.render(.{ .id = "compact", .count = planets.len, .viewport_width = 320, .viewport_height = 200, .stride = 320 + 16 }) .children({ for (0..planets.len) |i| slide(i, 320, 200); }); } /// Stops at the first and last slide instead of wrapping. pub fn bounded() void { Carousel.render(.{ .id = "bounded", .count = planets.len, .viewport_width = 420, .viewport_height = 240, .stride = 420 + 16, .loop = false }) .children({ for (0..planets.len) |i| slide(i, 420, 240); }); } /// No dots below. pub fn arrowsOnly() void { Carousel.render(.{ .id = "arrows-only", .count = planets.len, .viewport_width = 420, .viewport_height = 240, .stride = 420 + 16, .show_dots = false }) .children({ for (0..planets.len) |i| slide(i, 420, 240); }); } /// No arrows: swipe or use the dots. pub fn dotsOnly() void { Carousel.render(.{ .id = "dots-only", .count = planets.len, .viewport_width = 420, .viewport_height = 240, .stride = 420 + 16, .show_arrows = false }) .children({ for (0..planets.len) |i| slide(i, 420, 240); }); }

API Reference

Carousel.render options

id: []const u8, // stable id — required for state count: usize, // number of slides you render viewport_width: f32 = 480, // visible width (px) viewport_height: f32 = 280, // visible height (px) stride: f32 = 496, // slide width + gap (px) loop: bool = true, // wrap around at the ends show_arrows: bool = true, // prev / next buttons show_dots: bool = true, // dot indicators swipe_threshold: f32 = 60, // px drag needed to advance transition_ms: u32 = 350, // snap animation duration
