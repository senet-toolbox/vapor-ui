Custom Item Renderer

Use create() + Component() to render slides with your own component.

Installation

// In main.zig — initialize state map once at startup const Carousel = @import("components/Carousel.zig"); Carousel.new();

Usage

const Carousel = @import("components/Carousel.zig"); const items = [_]CarouselItem{ .{ .title = "Slide 1", .subtitle = "desc", .color = .hex("#3b82f6") }, .{ .title = "Slide 2", .subtitle = "desc", .color = .hex("#8b5cf6") }, // ... }; // Click + swipe carousel Carousel.render(.{ .id = "my-carousel", // required for state persistence .items = &items, .item_width = 480, .item_height = 280, });

API Reference

CarouselItem

pub const CarouselItem = struct { title: []const u8, subtitle: []const u8 = "", // optional label below title color: Color = .palette(.tint), // background color of slide };

Custom item renderer via create() + Component()

fn mySlide(item: CarouselItem, w: f32, h: f32) void { Stack() .width(.px(w)) .height(.px(h)) .background(item.color) .children({ Text(item.title).end(); }); } _ = Carousel.create(.{ .id = "custom", .items = &items, }).Component(mySlide, .{});

Carousel fields

id: []const u8 = "", // stable id — required for state items: []const CarouselItem, // slide data item_width: f32 = 480, // slide width (px) item_height: f32 = 280, // slide height (px) gap: f32 = 16, // gap between slides (px) loop: bool = true, // wrap around at the ends show_arrows: bool = true, // prev / next buttons show_dots: bool = true, // dot indicators swipe_threshold: f32 = 60, // px drag needed to advance transition_ms: u32 = 350, // snap animation duration
